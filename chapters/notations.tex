\section*{Notations and Conventions}

\textcolor{red}{THINGS TO INCLUDE HERE:
\begin{itemize}
    \item Feynman diagrams read left-to-right.
    \item Natural units.
    \item $(+,-,-,-)$ convention for metric.
    \item Four-vector vs. three-vector notation: $p,p^\mu$ vs. $\vec{v}$ (or $\mathbf{p}$). Slash notation.
    \item Group representation notation (if relevant). For instance $SU(3)$: $\mathbf{3}$ for fundamental, $\mathbf{8}$ for adjoint, and $\bar{\mathbf{3}}$ for dual.
    \item Fourier transform convention.
    \item Index notation ($\mu\ldots$ for four-vectors, $A,\dot{A}\ldots$ for Weyl spinors, etc.)
\end{itemize}
